Run the component Catalog

The repository is the source of truth; the crate is not yet published on crates.io.

git clone https://github.com/danieldear/volt-ui-kit
cd volt-ui-kit
cargo run --release --locked \
  --features native-gpu,native-desktop \
  --example catalog

Add it to your project

[dependencies]
volt-ui-kit = { git = "https://github.com/danieldear/volt-ui-kit", features = ["native-gpu"] }

For reproducible builds, pin a reviewed Git commit with rev and commit your application's lockfile.

src/main.rsState → view → event
use volt_ui_kit::{native::Application, ui::prelude::*};

#[derive(Default)]
struct App { count: u32 }

fn view(app: &App) -> Element<App> {
    vstack![
        text(format!("Count: {}", app.count))
            .font_size(24.),
        button("Add one")
            .key("add")
            .variant(ButtonVariant::Primary)
            .on_click(|app: &mut App| app.count += 1),
    ]
    .padding(24., Edge::All)
    .gap(12., Axis::Vertical)
    .into()
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    Application::new("Counter", App::default(), view).run()
}
